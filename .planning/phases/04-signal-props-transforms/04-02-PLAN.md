**IMPORTANT DESIGN NOTE on hoisted_qrls:** Since OXC Traverse doesn't allow storing arena-allocated AST nodes across enter/exit boundaries (the lifetimes don't work), store hoisted QRLs as serialized statement strings instead:

```rust
/// Stack of hoisted QRL declarations per function scope.
/// Each Vec contains (symbol_name, serialized_const_declaration) pairs.
hoisted_qrls: Vec<Vec<(String, String)>>,
```

The serialized form would be like: `const SegmentName = /* @__PURE__ */ qrl(i_hash, "SegmentName");`

When injecting into a block, parse these back into statements or insert them as raw text.

**Step 2: Add Traverse enter/exit hooks for loop constructs.**

Add these methods to the `impl<'a> Traverse<'a> for QwikTransform` block:

```rust
fn enter_for_statement(&mut self, node: &mut ForStatement<'a>, _ctx: &mut TraverseCtx<'a, ()>) {
    self.loop_depth += 1;
    // Extract iteration variable from init: for (let i = ...) or for (var i = ...)
    let iteration_vars = if let Some(ForStatementInit::VariableDeclaration(ref decl)) = node.init {
        decl.declarations.first()
            .and_then(|d| {
                if let BindingPattern::BindingIdentifier(ident) = &d.id {
                    Some(vec![ident.name.to_string()])
                } else {
                    None
                }
            })
            .unwrap_or_default()
    } else {
        Vec::new()
    };
    self.iteration_var_stack.push(iteration_vars);
}

fn exit_for_statement(&mut self, _node: &mut ForStatement<'a>, _ctx: &mut TraverseCtx<'a, ()>) {
    self.iteration_var_stack.pop();
    self.loop_depth -= 1;
}

fn enter_for_in_statement(&mut self, node: &mut ForInStatement<'a>, _ctx: &mut TraverseCtx<'a, ()>) {
    self.loop_depth += 1;
    let iteration_vars = match &node.left {
        ForStatementLeft::VariableDeclaration(decl) => {
            decl.declarations.first()
                .and_then(|d| {
                    if let BindingPattern::BindingIdentifier(ident) = &d.id {
                        Some(vec![ident.name.to_string()])
                    } else {
                        None
                    }
                })
                .unwrap_or_default()
        }
        ForStatementLeft::AssignmentTargetIdentifier(ident) => {
            vec![ident.name.to_string()]
        }
        _ => Vec::new(),
    };
    self.iteration_var_stack.push(iteration_vars);
}

fn exit_for_in_statement(&mut self, _node: &mut ForInStatement<'a>, _ctx: &mut TraverseCtx<'a, ()>) {
    self.iteration_var_stack.pop();
    self.loop_depth -= 1;
}

fn enter_for_of_statement(&mut self, node: &mut ForOfStatement<'a>, _ctx: &mut TraverseCtx<'a, ()>) {
    self.loop_depth += 1;
    let iteration_vars = match &node.left {
        ForStatementLeft::VariableDeclaration(decl) => {
            decl.declarations.first()
                .and_then(|d| {
                    if let BindingPattern::BindingIdentifier(ident) = &d.id {
                        Some(vec![ident.name.to_string()])
                    } else {
                        None
                    }
                })
                .unwrap_or_default()
        }
        ForStatementLeft::AssignmentTargetIdentifier(ident) => {
            vec![ident.name.to_string()]
        }
        _ => Vec::new(),
    };
    self.iteration_var_stack.push(iteration_vars);
}

fn exit_for_of_statement(&mut self, _node: &mut ForOfStatement<'a>, _ctx: &mut TraverseCtx<'a, ()>) {
    self.iteration_var_stack.pop();
    self.loop_depth -= 1;
}

fn enter_while_statement(&mut self, node: &mut WhileStatement<'a>, _ctx: &mut TraverseCtx<'a, ()>) {
    self.loop_depth += 1;
    // Extract iteration variable from test: while (i < ...) => extract "i"
    let iteration_vars = match &node.test {
        Expression::BinaryExpression(bin) => {
            if let Expression::Identifier(ident) = &bin.left {
                vec![ident.name.to_string()]
            } else {
                Vec::new()
            }
        }
        _ => Vec::new(),
    };
    self.iteration_var_stack.push(iteration_vars);
}

fn exit_while_statement(&mut self, _node: &mut WhileStatement<'a>, _ctx: &mut TraverseCtx<'a, ()>) {
    self.iteration_var_stack.pop();
    self.loop_depth -= 1;
}
```

// Check if this is an array iteration method call
if let Expression::StaticMemberExpression(member) = &call.callee {
    let method_name = member.property.name.as_str();
    if matches!(method_name,
        "map" | "filter" | "forEach" | "flatMap" | "some" | "every"
        | "find" | "findIndex" | "reduce" | "reduceRight"

        self.loop_depth += 1;
        self.in_callback = true;
        // Extract callback parameters as iteration variables
        if let Some(first_arg) = call.arguments.first() {
            let iteration_vars = match first_arg {
                Argument::ArrowFunctionExpression(arrow) => {
                    arrow.params.items.iter()
                        .filter_map(|p| {
                            if let BindingPattern::BindingIdentifier(ident) = &p.pattern {
                                Some(ident.name.to_string())
                            } else {
                                None
                            }
                        })
                        .collect()
                }
                Argument::FunctionExpression(func) => {
                    func.params.items.iter()
                        .filter_map(|p| {
                            if let BindingPattern::BindingIdentifier(ident) = &p.pattern {
                                Some(ident.name.to_string())
                            } else {
                                None
                            }
                        })
                        .collect()
                }
                _ => Vec::new(),
            };
            self.iteration_var_stack.push(iteration_vars);
        } else {
            self.iteration_var_stack.push(Vec::new());
        }
    }
}

**Part A: QRL hoisting (transform.rs)**

Add a method to QwikTransform:

```rust
/// If inside a loop, hoist the QRL call to a variable declaration.
/// Returns either a variable reference (if hoisted) or the original expression.
/// The symbol_name is used as both the variable name and deduplication key.
pub(crate) fn maybe_hoist_qrl(&mut self, qrl_call_code: &str, symbol_name: &str) -> Option<String> {
    if self.loop_depth == 0 || self.hoisted_qrls.is_empty() {
        return None; // Not in a loop, use inline
    }
    let depth = self.hoisted_qrls.len() - 1;
    // Check if already hoisted at this depth
    if self.hoisted_qrls[depth].iter().any(|(name, _)| name == symbol_name) {
        return Some(symbol_name.to_string()); // Reuse existing
    }
    // Store for later injection
    let const_decl = format!("const {} = {};", symbol_name, qrl_call_code);
    self.hoisted_qrls[depth].push((symbol_name.to_string(), const_decl));
    Some(symbol_name.to_string())
}
```

In `exit_expression` where QRL calls are built (the segment strategy path), check if hoisting is needed:

fn expr_uses_ident(expr: &Expression<'_>, name: &str) -> bool {
    match expr {
        Expression::Identifier(ident) => ident.name.as_str() == name,
        Expression::StaticMemberExpression(mem) => expr_uses_ident(&mem.object, name),
        Expression::ComputedMemberExpression(mem) => {
            expr_uses_ident(&mem.object, name) || expr_uses_ident(&mem.expression, name)
        }
        Expression::BinaryExpression(bin) => {
            expr_uses_ident(&bin.left, name) || expr_uses_ident(&bin.right, name)
        }
        Expression::CallExpression(call) => {
            expr_uses_ident(&call.callee, name)
                || call.arguments.iter().any(|a| match a {
                    Argument::SpreadElement(s) => expr_uses_ident(&s.argument, name),
                    _ => false, // simplified -- expand as needed
                })
        }
        Expression::ConditionalExpression(cond) => {
            expr_uses_ident(&cond.test, name)
                || expr_uses_ident(&cond.consequent, name)
                || expr_uses_ident(&cond.alternate, name)
        }
        Expression::ArrowFunctionExpression(arrow) => {
            arrow.body.statements.iter().any(|stmt| stmt_uses_ident(stmt, name))
        }
        // Add more expression types as needed
        _ => false,
    }
}
